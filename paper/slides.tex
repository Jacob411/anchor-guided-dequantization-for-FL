% ── 2. MOTIVATION ───────────────────────────────────────────────────────────
\begin{frame}{Federated Learning and the Quantization Problem}
\begin{columns}[T]
\column{0.54\textwidth}
\textbf{Federated learning:} train a model across distributed devices without sharing raw data.

\vspace{0.7em}
Each round:
\begin{enumerate}
  \item Server sends global model to clients
  \item Clients train locally, compute weight \emph{deltas}
  \item Clients send deltas back; server aggregates
\end{enumerate}

\vspace{0.7em}
\textbf{The problem:}\\
Not all clients can transmit FP32 updates.\\
Resource-constrained devices (IoT, phones) must \textbf{quantize} before sending.

\vspace{0.7em}
\begin{block}{Quantization bias}
  INT8/INT4 quantization introduces \textbf{systematic error} in every delta.\\
  FedAvg averages biased and clean updates together — the bias propagates into the global model.
\end{block}

\column{0.44\textwidth}

% ── 3. EXISTING APPROACH ────────────────────────────────────────────────────
\begin{frame}{Existing Approach: ProWD}
\begin{columns}[T]
\column{0.50\textwidth}
\textbf{ProWD} trains a server-side MLP to correct quantized deltas toward FP32.

\vspace{0.8em}
\textbf{How it works:}
\begin{enumerate}
  \item Each round: train MLP with MSE loss against the \emph{mean} anchor delta
  \item Apply MLP to constrained deltas before aggregation
  \item Reset MLP next round — no memory
\end{enumerate}

\vspace{0.8em}
\textbf{Two fundamental weaknesses:}
\begin{itemize}
  \item \textbf{Mean target}: all constrained clients corrected toward the same average, ignoring per-client heterogeneity
  \item \textbf{No replay buffer}: the MLP sees only this round's data — gradient estimates are noisy and unstable
\end{itemize}

\column{0.47\textwidth}
\vspace{0.5em}
\begin{block}{Result: ProWD is consistently worse}
  \vspace{0.3em}
  In every one of our experiments:\\[0.4em]
  \begin{center}
  \large AGD-Smart $>$ ProWD\\[0.2em]
  \small by \textbf{1.9 to 5.7 pp}
  \end{center}
\end{block}

\vspace{0.8em}
\begin{block}{Our goal}
  Fix both weaknesses — per-client matching and a growing replay buffer — while adding semantic distillation.
\end{block}
\end{columns}
\end{frame}

\column{0.46\textwidth}
\vspace{0.3em}
\textbf{Three improvements over ProWD:}

\vspace{0.5em}
\begin{block}{1. Residual CorrectionNet}
$f_\theta(x) = x + \text{Dec}(\text{Enc}(x))$\\
Zero-init decoder $\Rightarrow$ starts as identity, learns corrections gradually
\end{block}

\vspace{0.3em}
\begin{block}{2. Replay Buffer}
Accumulates $(q\text{-delta, FP32-target})$ pairs across \emph{all} rounds — stable gradient estimates every round
\end{block}

\vspace{0.3em}
\begin{block}{3. Smart Matching}
Each constrained client paired with its \emph{most cosine-similar} anchor — not the mean
\[
a^* = \arg\max_a \frac{\Delta w_j^q \cdot \Delta w_a^{fp}}{\|\Delta w_j^q\|\,\|\Delta w_a^{fp}\|}
\]
\end{block}
\end{columns}

% ── 5. TRAINING OBJECTIVE ───────────────────────────────────────────────────
\begin{frame}{Training Objective and Design Choices}
\begin{columns}[T]
\column{0.50\textwidth}
\textbf{Combined loss per round:}
\[
\mathcal{L} = \underbrace{\lambda_r \|f_\theta(\Delta w^q) - \Delta w^{fp}\|_2^2}_{\text{reconstruction}}
\;+\; \underbrace{\lambda_d\, D_{\mathrm{KL}}(p_{\text{anchor}} \| p_{\text{corrected}})}_{\text{distillation}}
\]

\vspace{0.3em}
\textbf{Distillation} ($\tau=4$): forces corrected updates to produce predictions aligned with anchor predictions — semantic anchoring, not just weight-space matching

\vspace{0.7em}
\textbf{Direction-preserving blend:}
\[
\hat{\Delta w}_j = \underbrace{\beta\, f_\theta(\Delta w_j^q)}_{\text{corrected}} + \underbrace{(1-\beta)\,\Delta w_j^q}_{\text{original}}
\]
$\beta=0.8$: keeps 20\% of client's own direction under non-IID data

\column{0.48\textwidth}
\vspace{0.3em}
\textbf{Adaptive warmup ($W=5$ rounds):}

\vspace{0.3em}
\includegraphics[width=\linewidth]{int2_warmup.png}

{\scriptsize CorrectionNet trains during warmup to fill the replay buffer, but correction is not applied until round 6. This prevents the untrained network from degrading early accuracy.}
\end{columns}
\end{frame}

% ── 8. NOISE SCALING THRESHOLD ──────────────────────────────────────────────
\begin{frame}{When Does Averaging Break Down?}
\begin{columns}[T]
\column{0.58\textwidth}
\includegraphics[width=\linewidth]{noise_scaling.png}

\column{0.40\textwidth}
\vspace{0.3em}
\textbf{Quantization levels:}\\[0.4em]
\begin{tabular}{lr}
INT8 & 255 levels \\
INT6 & 63 levels \\
INT4 & 15 levels \\
INT3 & 7 levels \\
\textbf{INT2} & \textbf{3 levels} \\
\end{tabular}

\vspace{0.8em}
\textbf{The threshold:}\\
At $\geq 7$ levels (INT3+): statistical averaging is sufficient — FedAvg wins

\vspace{0.5em}
At \textbf{3 levels (INT2)}: every non-zero gradient collapses to $\{-s, 0, +s\}$. Sign cancellation across clients destroys the signal. Averaging fails.

\vspace{0.6em}
\textbf{AGD-Smart beats ProWD at every bit-width} (by 1.8--10.9 pp). The replay buffer advantage is consistent regardless of noise level.
\end{columns}
\end{frame}

% ── 9. WHY ONLY INT2 ────────────────────────────────────────────────────────
\begin{frame}{Why Only INT2? The Threshold Explained}
\begin{columns}[T]
\column{0.50\textwidth}
\includegraphics[width=\linewidth]{vs_fedavg.png}

{\scriptsize Positive = AGD-Smart beats FedAvg.\\Only INT2 crosses zero.}

\column{0.48\textwidth}
\vspace{0.3em}
\textbf{INT2: only 3 quantization levels}
\[
\Delta w^q \;\in\; \{-s,\; 0,\; +s\}
\]

\textbf{Two problems at 3 levels:}
\begin{itemize}
  \item \textbf{Sign cancellation:} clients with balanced $\pm 1$ values average to $\approx 0$, destroying the update
  \item \textbf{Magnitude erasure:} large and small gradients get the same code — mean delta has wrong magnitudes
\end{itemize}

\vspace{0.5em}
\textbf{CorrectionNet escapes this} because it is a learned nonlinear mapping, not an average. It can learn: ``when client sends $\{-1,0,+1\}$, the true delta looks like \emph{this}.''

\vspace{0.5em}
\textbf{At INT3+ (7+ levels)}: enough resolution that averaging recovers the signal — CorrectionNet's overhead slightly hurts.
\end{columns}
\end{frame}

% ── 14. CONCLUSION ──────────────────────────────────────────────────────────
\begin{frame}{Conclusion}
\begin{columns}[T]
\column{0.52\textwidth}

\textbf{What we built:}\\
AGD-Smart — a server-side CorrectionNet for heterogeneous-quantization FL

\vspace{0.5em}
\begin{block}{Key design choices}
\begin{itemize}
  \item \textbf{Replay buffer} — stable learning across rounds, not just this round
  \item \textbf{Smart matching} — per-client, not mean anchor target
  \item \textbf{Distillation loss} — semantic alignment, not just weight-space MSE
  \item \textbf{Adaptive warmup} — buffer fills before correction fires
\end{itemize}
\end{block}

\vspace{0.5em}
\begin{block}{Core finding}
There is a \textbf{sharp threshold at 3 quantization levels (INT2)} below which statistical averaging fails. AGD-Smart is specifically built for this regime — and is the only method that beats FedAvg there.
\end{block}
